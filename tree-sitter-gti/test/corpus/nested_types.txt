public:
  struct Network {
    int port;
    Network(int value) : port(value) {}
    class Credentials { public: Credentials() {} };
    Network() : port(0) {}
  };
  [[no_transfer]] class Logging { public: Logging() {} };
  Config() : network(80) {}
  ~Config() {}
  Network network;
};
Config::Network make() { return Config::Network(443); }
