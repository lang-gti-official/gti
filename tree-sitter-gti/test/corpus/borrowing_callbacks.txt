  auto increment = [](mut int& value) -> mut int& { value += 1; return value; };
  auto observe = [](mut int& value) -> int& { value += 1; return value; };
  auto relay = [](mut int& value) -> mut int& {
    mut int& alias = value;
    return alias;
  };
  return 0;
}
